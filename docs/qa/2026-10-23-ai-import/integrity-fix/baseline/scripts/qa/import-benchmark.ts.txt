/** Explicit paid QA only. Synthetic fixtures, serial calls, no DB/storage, $8 admission ceiling. */
import fs from 'node:fs';
import path from 'node:path';
import {createHash} from 'node:crypto';
import Anthropic from '@anthropic-ai/sdk';
import { betaZodOutputFormat } from '@anthropic-ai/sdk/helpers/beta/zod';
import { prepareImportFile, toContentBlocks } from '../../lib/import/files';
import { importProposalSchema, IMPORT_CHANNELS } from '../../lib/import/proposal';

const ROOT=path.resolve('docs/qa/2026-10-23-ai-import');
const output=path.join(ROOT,'benchmark'); fs.mkdirSync(output,{recursive:true});
const ledgerFile=path.join(output,'ledger.json');
const ledger: any[]=fs.existsSync(ledgerFile)?JSON.parse(fs.readFileSync(ledgerFile,'utf8')):[];
const prices: Record<string,[number,number]>={'claude-opus-5-5':[4,20],'claude-sonnet-5-5':[2,10],'claude-haiku-5-5':[.1,.5]};
// Only the explicitly authorized Anthropic key is loaded, in memory, never logged/saved.
let key=process.env.ANTHROPIC_API_KEY;
if (!key && process.env.QA_KEY_DIRECTORY) {
  for (const name of ['.env','.env.local']) {
    const file=path.join(process.env.QA_KEY_DIRECTORY,name);
    if (!fs.existsSync(file)) continue;
    for (const line of fs.readFileSync(file,'utf8').split('\n')) {
      const match=line.match(/^\s*(?:export\s+)?ANTHROPIC_API_KEY\s*=\s*(.*?)\s*$/);
      if (match?.[1]) key=match[1].replace(/^(["'])(.*)\1$/,'$2');
    }
  }
}
if (!key) throw new Error('Anthropic key unavailable; no calls made.');
const api=new Anthropic({apiKey:key,baseURL:'https://api.anthropic.com',maxRetries:0,timeout:240000});
const prompt=fs.readFileSync('lib/import/claude.ts','utf8').match(/const SYSTEM_PROMPT = `([\s\S]*?)`;/)?.[1];
if(!prompt) throw new Error('Cannot locate production prompt');
const save=()=>fs.writeFileSync(ledgerFile,JSON.stringify(ledger,null,2));
async function main(){
  if(ledger.some(x=>x.status==='in_flight'||x.status==='error_or_uncertain'))throw new Error('Resolve existing uncertain reservation before more paid calls');
  const hard=process.env.QA_SUITE==='hard';
  const fixtures=path.join(ROOT,'fixtures',hard?'supplement':'');
  const clarity=process.env.QA_PROMPT_REVISION==='clarity';
  const cases=fs.readdirSync(fixtures).filter(n=>hard?/^(07|08|09|10|11)-/.test(n):/^0[1-6]-/.test(n)).filter(n=>!clarity||['07-pdf-raster','08-sem-manifesto','11-instrucao'].includes(n));
  for(const fixtureId of cases){
    const caseId=(clarity?'clarity--':hard?'hard--':'')+fixtureId;
    const dir=path.join(fixtures,fixtureId);
    const gold=JSON.parse(fs.readFileSync(path.join(dir,'expected.json'),'utf8'));
    const start=performance.now();
    const files=await Promise.all(gold.documents.map(async (n:string)=>prepareImportFile(n,new Uint8Array(fs.readFileSync(path.join(dir,n))))));
    const extractionMs=performance.now()-start;
    const blocks:any[]=toContentBlocks(files);
    for(const name of gold.media){
      if(name.endsWith('.mp4'))continue;
      blocks.push({type:'text',text:`Mídia: ${name}`},{type:'image',source:{type:'base64',media_type:'image/png',data:fs.readFileSync(path.join(dir,name)).toString('base64')}});
    }
    blocks.push({type:'text',text:`Cliente: ${gold.client}\nData de referência (hoje): ${gold.referenceDate}\nCanais permitidos: ${IMPORT_CHANNELS.join(', ')}\n\nArquivos de mídia enviados:\n${gold.media.map((n:string)=>`- ${n} (${n.endsWith('.mp4')?'vídeo':'imagem'})`).join('\n')||'(nenhuma)'}\n\nMigre o calendário a partir dos arquivos acima.`});
    const max_tokens=caseId==='06-limite'?64000:Math.max(6000,(gold.expectedPosts??0)*500+2000);
    for(const model of (hard?['claude-haiku-5-5','claude-sonnet-5-5']:Object.keys(prices))){
      if(ledger.some(x=>x.caseId===caseId&&x.requestedModel===model))continue;
      const count=await api.messages.countTokens({model,system:prompt!,messages:[{role:'user',content:blocks}]});
      const rate=model==='claude-haiku-5-5'&&count.input_tokens>100000?[.5,2.5]:prices[model];
      // Include generous schema overhead in the reservation; no fallback to a costlier model.
      const reservedUSD=((count.input_tokens+5000)*rate[0]+max_tokens*rate[1])/1e6;
      const spent=ledger.reduce((s,x)=>s+(x.costUSD??x.reservedUSD),0);
      if(spent+reservedUSD>8){console.log('Budget admission refused',caseId,model);return;}
      const row:any={caseId,requestedModel:model,countedInputTokens:count.input_tokens,maxOutputTokens:max_tokens,reservedUSD,extractionMs,status:'in_flight',startedAt:new Date().toISOString(),schemaCache:'uncontrolled; first invocation may compile schema',promptSha256:createHash('sha256').update(prompt!).digest('hex'),inputSha256:createHash('sha256').update(JSON.stringify(blocks)).digest('hex')};
      ledger.push(row);save();
      console.log('Start',caseId,model,'reserved',reservedUSD.toFixed(4),'cumulative',spent.toFixed(4));
      const t=performance.now();
      try{
        const result=await api.beta.messages.stream({model,max_tokens,output_config:{effort:'medium',format:betaZodOutputFormat(importProposalSchema)},system:prompt!,messages:[{role:'user',content:blocks}]}).finalMessage();
        row.latencyMs=performance.now()-t;row.actualModel=result.model;row.stopReason=result.stop_reason;row.usage=result.usage;
        row.costUSD=((result.usage.input_tokens)*rate[0]+result.usage.output_tokens*rate[1]+(result.usage.cache_creation_input_tokens??0)*rate[0]*2+(result.usage.cache_read_input_tokens??0)*rate[0])/1e6;
        row.status=result.stop_reason==='end_turn'&&result.parsed_output?'complete':'incomplete';
        if(result.parsed_output){
          const proposal:any=result.parsed_output;
          fs.writeFileSync(path.join(output,`${caseId}-${model}.json`),JSON.stringify(proposal,null,2));
          let exactCopy=0,exactSchedule=0,exactChannel=0,exactMedia=0,correct=0;
          const differences:any[]=[];
          for(const expected of gold.proposal?.posts??[]){
            const found=proposal.posts.find((p:any)=>p.title===expected.title&&p.channel===expected.channel);
            if(!found){differences.push({title:expected.title,issue:'missing or renamed'});continue;}
            const copy=found.copyText===expected.copyText;
            const schedule=found.date===expected.date&&found.time===expected.time;
            const media=JSON.stringify(found.media)===JSON.stringify(expected.media);
            if(copy)exactCopy++;if(schedule)exactSchedule++;if(media)exactMedia++;exactChannel++;
            if(copy&&schedule&&media&&found.format===expected.format)correct++;
            else differences.push({title:expected.title,copy,schedule,media,format:found.format===expected.format});
          }
          row.quality={expected:gold.expectedPosts,found:proposal.posts.length,exactCopy,exactSchedule,exactChannel,exactMedia,fullyCorrect:correct,costPerFullyCorrect:correct?row.costUSD/correct:null,differences};
        }
      }catch(e:any){row.latencyMs=performance.now()-t;row.status='error_or_uncertain';row.errorType=e?.name;row.httpStatus=e?.status??null;/* no raw provider errors or headers */}
      save();console.log('End',caseId,model,row.status,'seconds',(row.latencyMs/1000).toFixed(1),'cost/reserve',(row.costUSD??reservedUSD).toFixed(4));
      if(row.status==='error_or_uncertain'){console.log('Stopped after uncertain outcome; reservation retained.');return;}
    }
  }
}
main().catch(e=>{console.error('Benchmark stopped:',e?.name);process.exitCode=1;});
